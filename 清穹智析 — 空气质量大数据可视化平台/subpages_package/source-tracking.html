# ========== Python 爬虫示例 ==========

import requests
import pandas as pd

# ---------- 1. 从 aqicn.org API 获取实时AQI (需token) ----------
def fetch_aqicn(city="beijing", token="your_token"):
    """城市名拼音，如 beijing, shanghai"""
    url = f"http://api.waqi.info/feed/{city}/?token={token}"
    resp = requests.get(url).json()
    if resp['status'] == 'ok':
        data = resp['data']
        return {
            'city': data['city']['name'],
            'aqi': data['aqi'],
            'pm25': data['iaqi'].get('pm25', {}).get('v'),
            'pm10': data['iaqi'].get('pm10', {}).get('v'),
            'time': data['time']['s']
        }
    return None
