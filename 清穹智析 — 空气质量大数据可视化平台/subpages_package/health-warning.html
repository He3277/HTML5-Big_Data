查看数据爬虫示例 (Python)

数据源: https://aqicn.org (世界空气质量指数) 或 中国环境监测总站

import requests
from bs4 import BeautifulSoup

# 示例：爬取aqicn.org某城市实时数据 (实际使用时请遵守robots.txt)
city = "beijing"
url = f"https://aqicn.org/city/{city}/"
headers = {'User-Agent': 'Mozilla/5.0'}
resp = requests.get(url, headers=headers)
soup = BeautifulSoup(resp.text, 'html.parser')

# 解析AQI值 (需根据页面结构调整)
aqi_val = soup.find('div', class_='aqivalue').text.strip()
print(f"{city} AQI: {aqi_val}")
